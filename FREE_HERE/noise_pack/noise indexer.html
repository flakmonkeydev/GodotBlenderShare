Select an RGBA packed noise texture to extract individual noise channels by ID.

Godot Shader: get_noise_by_id()

// Fetch scalar noise by ID from a packed 4-channel RGBA texture
float get_noise_by_id(sampler2D noise_tex, vec2 uv, int id) {
    vec4 n = texture(noise_tex, uv);
    switch (id) {
        case 0: return n.r;
        case 1: return n.g;
        case 2: return n.b;
        case 3: return n.a;
        default: return 0.0;
    }
}
